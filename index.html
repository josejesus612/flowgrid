03import { useState, useEffect } from 'react'
04import { supabase } from '@/lib/supabase'
05
06export function useEquipment(orgId: string) {
07  const [certs, setCerts] = useState([]);
08  const [loading, setLoading] = useState(true);
09  useEffect(() => {
10    async function load() {
11      const { data } = await supabase
12        .from('certificates')
13        .select('*')
14        .eq('org_id', orgId)
15        .order('expiry_date');
16      setCerts(data ?? []);
17      setLoading(false);
18    }
19    load();
20  }, [orgId]);
21  return { certs, loading };
22}
